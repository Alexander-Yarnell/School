const folderPath = "01 Journals/00 Daily Notes";
const templatePath = "04 Templates/00 Journals/Daily Journal - Template.md";

const current = moment(dv.current().file.name, "YYYY-MM-DD", true);

if (current.isValid()) {
  const yesterday = current.clone().subtract(1, "days").format("YYYY-MM-DD");
  const tomorrow = current.clone().add(1, "days").format("YYYY-MM-DD");

  const container = dv.el("div", "", { cls: "daily-nav" });

  async function openOrCreate(dateStr) {
    const fullPath = `${folderPath}/${dateStr}.md`;
    const existing = app.vault.getAbstractFileByPath(fullPath);

    if (existing) {
      app.workspace.openLinkText(dateStr, "", false);
      return;
    }

    const templater = app.plugins.plugins["templater-obsidian"]?.templater;
    if (!templater) {
      new Notice("Templater plugin not found.");
      return;
    }

    const templateFile = app.vault.getAbstractFileByPath(templatePath);
    const folder = app.vault.getAbstractFileByPath(folderPath);

    if (!templateFile || !folder) {
      new Notice("Template or folder path is wrong — check the script.");
      return;
    }

    await templater.create_new_note_from_template(templateFile, folder, dateStr, true);
  }

  const yesterdayLink = container.createEl("a", { text: "← Yesterday", href: "#" });
  yesterdayLink.addEventListener("click", (e) => { e.preventDefault(); openOrCreate(yesterday); });

  const tomorrowLink = container.createEl("a", { text: "Tomorrow →", href: "#" });
  tomorrowLink.addEventListener("click", (e) => { e.preventDefault(); openOrCreate(tomorrow); });

} else {
  dv.paragraph("⚠️ Filename isn't in YYYY-MM-DD format — nav buttons need that to calculate dates.");
}
